test('exact square-root rounding and ceilings retain rational boundaries beyond binary resolution',()=>{
 for(const k of [0,1,10,20,1000,Number.MAX_SAFE_INTEGER-1]){
  const square=exactMultiply(k,k);
  // For very large k build 2*k+1 without a native integer precision loss.
  const exactMidpoint=exactMultiply(exactDivide({numerator:2n*BigInt(k)+1n,denominator:1n},2),exactDivide({numerator:2n*BigInt(k)+1n,denominator:1n},2));
  const tiny={numerator:1n,denominator:10n**100n};
  assert.equal(exactSqrtCeil(square),k);assert.equal(exactSqrtCeil(exactSubtract(square,{numerator:-1n,denominator:10n**100n})),k+1);
  if(k>0)assert.equal(exactSqrtCeil(exactSubtract(square,tiny)),k);
  assert.equal(exactSqrtRound(exactSubtract(exactMidpoint,tiny)),k);
  assert.equal(exactSqrtRound(exactMidpoint),k+1);
  assert.equal(exactSqrtRound(exactSubtract(exactMidpoint,{numerator:-1n,denominator:10n**100n})),k+1);
 }
 assert.throws(()=>exactSqrtRound(-1),RangeError);assert.throws(()=>exactSqrtCeil(-1),RangeError);
 assert.throws(()=>exactSqrtRound(1e100),RangeError);assert.throws(()=>exactSqrtCeil(Infinity),TypeError);
 // Measured-outline area can be more precise than any Number. No display
 // conversion may decide its span count: side exceeds ten by far less than ULP.
 const area={numerator:100n*10n**100n+1n,denominator:10n**100n};
 assert.equal(exactToNumber(controlJointLength({dimensionMethod:'measured_outline'},area,10).quantity),20);
});
